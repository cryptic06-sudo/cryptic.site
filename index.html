let socket = null; function connectDeriv() { // 1089 is Deriv's free testing App ID const appId = 1089; // Open connection socket = new WebSocket(`wss://ws.derivws.com/websockets/v3?app_id=${appId}`); socket.onopen = function() { // Find status element and change text to Connected const statusText = document.getElementById("status") || document.querySelector("p"); if (statusText) statusText.innerText = "Connected"; // Start receiving live digits for Volatility 100 Index socket.send(JSON.stringify({ ticks: "R_100", subscribe: 1 })); }; socket.onmessage = function(event) { const data = JSON.parse(event.data); console.log("Data received:", data); }; socket.onerror = function(err) { alert("Connection Error. Check your internet or App ID."); }; socket.onclose = function() { const statusText = document.getElementById("status") || document.querySelector("p"); if (statusText) statusText.innerText = "Disconnected"; }; }
